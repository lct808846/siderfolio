client=httpx.Client(base_url=origin,timeout=180,follow_redirects=False)
def request(method,path,**kwargs):
 response=client.request(method,path,**kwargs)
 if not response.is_success:
  try:code=response.json().get('error',{}).get('code','http_error')
  except Exception:code='non_json_error'
  raise RuntimeError(f'{method} {path.split("/")[-1]}: {response.status_code} {code}')
 return response
def check(name,fn):
 start=time.monotonic()
 try:
  detail=fn();result['checks'][name]={'passed':True,'seconds':round(time.monotonic()-start,2),'detail':detail}
 except Exception as e:
  result['checks'][name]={'passed':False,'seconds':round(time.monotonic()-start,2),'error':str(e)[:220]}
 (record/'authenticated-live-first.json').write_text(json.dumps(result,indent=2),encoding='utf8')
 print(name,result['checks'][name]['passed'],flush=True)

def plotting(engine):
 status=request('GET','/api/v1/setup/plotting').json()
 assert status['storage_ready'] and status['execution_configured'], 'Executor unavailable'
 plan,run=str(uuid4()),str(uuid4())
 p=request('PUT',base+'/plotting/plans/'+plan,json={'dataset_id':dataset,'model_id':model,'engine':engine,'question':'Plot a scatter of x versus y from the synthetic dataset; no scientific inference.','confirm_external_data':True}).json()
 assert p['state']=='completed', 'Plan failed'
 value=request('PUT',base+'/plotting/runs/'+run,json={'plan_id':plan,'confirm_execute':True}).json()
 assert value['state']=='completed', 'Sandbox execution did not complete'
 figure=request('GET',base+'/plotting/runs/'+run+'/files/figure.png').content
 assert figure.startswith(b'\x89PNG\r\n\x1a\n')
 return {'png_sha256':hashlib.sha256(figure).hexdigest(),'real_internal_executor':True,'engine':engine}

def chat(family,message):
 conversation=request('POST','/api/v1/conversations',json={'title':'Public '+family+' acceptance'}).json()
 value=request('POST','/api/v1/conversations/'+conversation['id']+'/turns',json={'message':message,'client_request_id':str(uuid4()),'expected_revision':0,'model':model}).json()
 if value.get('execution_status')!='completed':raise RuntimeError('Turn failed: '+str((value.get('error') or {}).get('code','unknown')))
 response=value.get('response') or {}
 results=response.get({'simbad':'simbad_results','ads':'literature_results','gaia':'gaia_results'}[family],[])
 if not results:
  raise RuntimeError('No successful '+family+' result; error codes: '+str([v.get('code') for v in response.get('errors',[])]))
 return {'completed':True,'result_count':len(results)}
